# Dash research dashboard dependencies, pinned. See ../requirements.txt for
# the pinning rationale.
#
# blpapi (DATA_SOURCE="bloomberg") is deliberately NOT listed here: it isn't
# on public PyPI (Bloomberg distributes it from their own package index), so
# `pip install -r requirements.txt` would fail the build on any standard
# cloud platform if it were. It's also local-only by nature — it talks to a
# running Bloomberg Terminal, which no cloud container has. Install it by
# hand (`pip install --index-url https://blpapi.bloomberg.com/repository/releases/python/simple/ blpapi`)
# only if running this locally with DATA_SOURCE="bloomberg".
yfinance==1.7.0
dash==4.4.1
dash-bootstrap-components==2.0.4
plotly==6.9.0
pandas==2.3.3

# numpy is the one dependency that cannot take a single pin: 2.5.x does not
# support Python 3.11, which is what runs locally, while the deployment VM is
# on 3.14. Markers keep both environments exactly reproducible rather than
# forcing one of them onto an untested version.
numpy==2.4.6; python_version < "3.12"
numpy==2.5.2; python_version >= "3.12"
